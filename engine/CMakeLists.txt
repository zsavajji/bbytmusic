cmake_minimum_required(VERSION 3.20)
project(bbmusic LANGUAGES C CXX VERSION 1.0.0)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(CMAKE_CROSSCOMPILING OR QNX)
  set(BBM_BB10 ON)
else()
  set(BBM_BB10 OFF)
endif()

option(BBM_SANITIZE "ASan + UBSan on the host build" ON)
add_compile_options(-Wall -Wextra)
if(BBM_SANITIZE AND NOT BBM_BB10)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

# ---- third-party libraries, as one target each ----------------------------------------------
if(BBM_BB10)
  # zlib by its .a, BY PATH, never -lz: the phone's libz.so.2 exports a libgcc unwinder that
  # hijacks C++ exceptions (phase0.md, gate 0.3). Order matters: curl, wolfssl, opus, libz.a.
  set(BBM_PREFIX /opt/bbmusic)
  add_library(bbm_curl INTERFACE)
  target_include_directories(bbm_curl INTERFACE ${BBM_PREFIX}/include)
  target_compile_definitions(bbm_curl INTERFACE CURL_STATICLIB)
  target_link_libraries(bbm_curl INTERFACE ${BBM_PREFIX}/lib/libcurl.a ${BBM_PREFIX}/lib/libwolfssl.a)
  add_library(bbm_opus INTERFACE)
  target_include_directories(bbm_opus INTERFACE ${BBM_PREFIX}/include)
  target_link_libraries(bbm_opus INTERFACE ${BBM_PREFIX}/lib/libopus.a)
  # libz.a and -lsocket come after curl/wolfssl/opus on the link line; bbm_syslibs is linked last.
  add_library(bbm_syslibs INTERFACE)
  target_link_libraries(bbm_syslibs INTERFACE
    /opt/bb10-sdk/target/qnx6/armle-v7/usr/lib/libz.a -lsocket)
  target_link_libraries(bbm_curl INTERFACE bbm_syslibs)
  # store uses the phone's own SQLite (3.7.14.1, which has WAL): -lsqlite3, no unwinder in it.
else()
  find_package(CURL REQUIRED)
  add_library(bbm_curl INTERFACE)
  target_link_libraries(bbm_curl INTERFACE CURL::libcurl)
  find_package(SQLite3 REQUIRED)
  list(APPEND CMAKE_PREFIX_PATH /opt/homebrew/opt/opus)
  set(ENV{PKG_CONFIG_PATH} "$ENV{PKG_CONFIG_PATH}:/opt/homebrew/opt/opus/lib/pkgconfig:/opt/homebrew/lib/pkgconfig:/usr/local/lib/pkgconfig")
  find_package(PkgConfig QUIET)
  add_library(bbm_opus INTERFACE)
  if(PkgConfig_FOUND)
    pkg_check_modules(OPUS QUIET IMPORTED_TARGET opus)
  endif()
  # Sources include <opus/opus.h>, as on BB10: the prefix's include dir, not .../include/opus.
  if(TARGET PkgConfig::OPUS)
    target_link_libraries(bbm_opus INTERFACE PkgConfig::OPUS)
    target_include_directories(bbm_opus INTERFACE /opt/homebrew/opt/opus/include)
  else()
    target_include_directories(bbm_opus INTERFACE /opt/homebrew/opt/opus/include)
    target_link_libraries(bbm_opus INTERFACE /opt/homebrew/opt/opus/lib/libopus.dylib)
  endif()
endif()

# ---- one library per module -----------------------------------------------------------------
set(BBM_MODULES util net proto stream media audio innertube player store platform server engine)
set(deps_util)
set(deps_net util)
set(deps_proto)
set(deps_stream proto net util)
set(deps_media util)
set(deps_audio util)
set(deps_innertube net util)
set(deps_player stream media audio)
set(deps_store util)
set(deps_platform util)   # BB10 only: its sources are empty on the host (#ifdef __QNX__)
set(deps_server util net proto stream media audio innertube player store platform)
set(deps_engine util net proto stream media audio innertube player store platform server)   # the wiring

set(YYJSON_DIR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/yyjson)

foreach(m ${BBM_MODULES})
  file(GLOB srcs CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/${m}/*.cpp)
  if(m STREQUAL "util")
    list(APPEND srcs ${YYJSON_DIR}/yyjson.c)
    set_source_files_properties(${YYJSON_DIR}/yyjson.c PROPERTIES COMPILE_OPTIONS "-w")
  endif()
  if(srcs)
    add_library(${m} STATIC ${srcs})
    set(vis PUBLIC)
  else()
    add_library(${m} INTERFACE)  # no sources yet: headers only
    set(vis INTERFACE)
  endif()
  target_include_directories(${m} ${vis} ${CMAKE_CURRENT_SOURCE_DIR}/src)
  foreach(d ${deps_${m}})
    if(TARGET ${d})
      target_link_libraries(${m} ${vis} ${d})
    endif()
  endforeach()
  if(m STREQUAL "util")
    target_include_directories(util PRIVATE ${YYJSON_DIR})
  elseif(m STREQUAL "net")
    target_link_libraries(net ${vis} bbm_curl)
  elseif(m STREQUAL "media")
    target_link_libraries(media ${vis} bbm_opus)
  elseif(m STREQUAL "audio" AND BBM_BB10)
    target_link_libraries(audio ${vis} -lasound -laudio_manager)   # qsa_sink; neither exports an unwinder
  elseif(m STREQUAL "platform" AND BBM_BB10)
    target_link_libraries(platform ${vis} -lbps -laudio_manager)   # bluetooth_route()
  elseif(m STREQUAL "store" AND BBM_BB10)
    target_link_libraries(store ${vis} -lsqlite3)
  elseif(m STREQUAL "store")
    if(TARGET SQLite3::SQLite3)
      target_link_libraries(store ${vis} SQLite3::SQLite3)
    else()
      target_link_libraries(store ${vis} SQLite::SQLite3)  # CMake < 3.35
    endif()
  endif()
endforeach()

# ---- the daemon -----------------------------------------------------------------------------
add_executable(bbmusicd src/main.cpp)
target_compile_definitions(bbmusicd PRIVATE BBM_VERSION="${PROJECT_VERSION}")
target_link_libraries(bbmusicd PRIVATE engine)

# The phone's own probe: the real media, audio and platform modules, run from a shell on the Q10.
if(BBM_BB10 AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tools/bb10_probe.cpp)
  add_executable(bb10_probe tools/bb10_probe.cpp)
  target_link_libraries(bb10_probe PRIVATE media audio platform util)
endif()

# ---- tests: tests/<module>/*_test.cpp, one program each ------------------------------------
if(NOT BBM_BB10)
  enable_testing()
  foreach(m ${BBM_MODULES})
    add_custom_target(${m}_tests)
    file(GLOB tests CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tests/${m}/*_test.cpp)
    foreach(t ${tests})
      get_filename_component(file ${t} NAME_WE)       # json_test, http_live_test
      string(REGEX REPLACE "_test$" "" name ${file})  # json, http_live
      set(exe ${m}_${name}_test)
      add_executable(${exe} ${t})
      target_include_directories(${exe} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
      target_compile_definitions(${exe} PRIVATE
        BBM_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
      target_link_libraries(${exe} PRIVATE ${m})
      add_dependencies(${m}_tests ${exe})
      add_test(NAME ${m}.${name} COMMAND ${exe})
      if(file MATCHES "_live_test$")
        set_tests_properties(${m}.${name} PROPERTIES LABELS live)
      endif()
    endforeach()
  endforeach()
endif()
